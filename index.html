Integrasi Google Apps Script (GAS)

Aplikasi ini dapat dihubungkan langsung ke Google Sheets dan Google Drive menggunakan Google Apps Script agar semua data tersimpan di Cloud dan PDF tergenerate otomatis di Drive.

Langkah Pemasangan Script GAS:

  1. Buka Google Spreadsheet baru di Google Drive Anda.
  2. Buka menu Ekstensi > Apps Script.
  3. Hapus kode bawaan, lalu Salin / Paste kode di bawah ini ke dalam Code.gs.
  4. Klik Terapkan (Deploy) > Terapkan sebagai Aplikasi Web (Web App).
  5. Akses set ke: Anyone (Siapa Saja). Simpan URL Web App tersebut.
// =============================================
// GOOGLE APPS SCRIPT BACKEND (Code.gs)
// Aplikasi Perjalanan Dinas Kecamatan
// =============================================

function doGet(e) {
  return HtmlService.createHtmlOutput("API Perjalanan Dinas Aktif.");
}

function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    
    // Auto Create Header jika Sheet Masih Kosong
    if (sheet.getLastRow() === 0) {
      sheet.appendRow([
        "Timestamp", "No Surat Tugas", "No SPPD", "Pelaksana", 
        "NIP Pelaksana", "Tujuan", "Tgl Mulai", "Tgl Selesai", 
        "Lama Hari", "Total Biaya", "Status"
      ]);
    }
    
    // Simpan Data
    sheet.appendRow([
      new Date(),
      data.noSuratTugas,
      data.noSPPD,
      data.namaPelaksana,
      data.nipPelaksana,
      data.tempatTujuan,
      data.tglMulai,
      data.tglSelesai,
      data.lamaHari,
      data.totalBiaya,
      "Tersimpan"
    ]);
    
    return ContentService.createTextOutput(JSON.stringify({
      "result": "success",
      "message": "Data Perjalanan Dinas Berhasil Disimpan ke Google Sheets!"
    })).setMimeType(ContentService.MimeType.JSON);
    
  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({
      "result": "error",
      "message": error.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  }
}
